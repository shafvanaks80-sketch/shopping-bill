def shopping_bill():
    products = []

    n = int(input("Enter the number of products: "))

    for i in range(n):
        print("\nProduct", i + 1)

        name = input("Enter product name: ")

        while True:
            try:
                price = float(input("Enter unit price: "))
                if price > 0:
                    break
                else:
                    print("Price must be greater than zero!")
            except ValueError:
                print("Please enter a valid price!")

        while True:
            try:
                quantity = int(input("Enter quantity: "))
                if quantity > 0:
                    break
                else:
                    print("Quantity must be a positive whole number!")
            except ValueError:
                print("Please enter a valid quantity!")

        subtotal = price * quantity

        products.append({
            "name": name,
            "price": price,
            "quantity": quantity,
            "subtotal": subtotal
        })

    # Calculate total
    total = sum(p["subtotal"] for p in products)

    # Calculate discount
    if total >= 2000:
        discount = total * 0.10
    else:
        discount = 0

    # Calculate tax
    amount = total - discount
    tax = amount * 0.05
    final_amount = amount + tax

    # Find most expensive product
    expensive = max(products, key=lambda p: p["price"])

    # Print bill
    print("\n========== SHOPPING BILL ==========")
    print(f"{'Product':<15}{'Price':>10}{'Qty':>8}{'Subtotal':>12}")
    print("-" * 45)

    for p in products:
        print(f"{p['name']:<15}{p['price']:>10.2f}"
              f"{p['quantity']:>8}{p['subtotal']:>12.2f}")

    print("-" * 45)
    print(f"Total Before Discount: Rs. {total:.2f}")
    print(f"Discount:              Rs. {discount:.2f}")
    print(f"Tax (5%):              Rs. {tax:.2f}")
    print(f"Final Payable Amount:  Rs. {final_amount:.2f}")
    print("Most Expensive Product:", expensive["name"])
    print("===================================")


shopping_bill()
